+#!/usr/bin/env node
+/**
+ * Генерация sitemap.xml на основе реального времени изменения файлов/статей.
+ *
+ * Запускается:
+ *   - Вручную: `node scripts/generate-sitemap.mjs`
+ *   - Автоматически в postbuild (до или после prerender)
+ *
+ * Что делает:
+ *   1. Для каждой основной страницы (/about, /skills и т.д.) находит
+ *      дату последнего коммита соответствующего JSX-файла через git log.
+ *   2. Для каждой статьи (/publications/:slug) берёт дату из статьи
+ *      или дату коммита/изменения.
+ *   3. Записывает sitemap.xml в public/sitemap.xml (и в build/sitemap.xml, если build/ существует).
+ */
+import fs from 'node:fs';
+import path from 'node:path';
+import { execSync } from 'node:child_process';
+import { fileURLToPath } from 'node:url';
+import { articleSlugs } from './routes.mjs';
+
+const __dirname = path.dirname(fileURLToPath(import.meta.url));
+const ROOT = path.resolve(__dirname, '..');
+const PUBLIC_DIR = path.join(ROOT, 'public');
+const BUILD_DIR = path.join(ROOT, 'build');
+const BASE_URL = 'https://alex-gsm.at';
+
+function getGitLastmod(relPath) {
+  try {
+    const fullPath = path.join(ROOT, relPath);
+    if (!fs.existsSync(fullPath)) return null;
+    const out = execSync(`git log -1 --format=%cI -- "${fullPath}"`, {
+      cwd: ROOT,
+      encoding: 'utf8',
+    }).trim();
+    if (out) return out.split('T')[0];
+  } catch {
+    /* fallback to file mtime */
+  }
+  try {
+    const fullPath = path.join(ROOT, relPath);
+    const stat = fs.statSync(fullPath);
+    return stat.mtime.toISOString().split('T')[0];
+  } catch {
+    return new Date().toISOString().split('T')[0];
+  }
+}
+
+// Загружаем данные статей
+const articlesFile = path.join(ROOT, 'src', 'data', 'articles.js');
+let articles = [];
+try {
+  const content = fs.readFileSync(articlesFile, 'utf8');
+  const jsonMatch = content.match(/export const articles = (\[[\s\S]*?\]);/);
+  if (jsonMatch) {
+    articles = JSON.parse(jsonMatch[1]);

+} catch (e) {

+  fs.writeFileSync(buildSitemap, xml, 'utf8');

+#!/usr/bin/env node
+/**
+ * SEO-аудит длины и уникальности <title> и <meta name="description">.
+ *
+ * Запускается автоматически в postbuild (после пререндера) — проверяет
+ * итоговый HTML в build/. Можно запускать вручную: `node scripts/seo-audit.mjs`
+ * (если build/ нет — проверяет исходники: src/data/seoMeta.js и articles.js).
+ *
+ * Правила: title ≤ 70 символов; description 25–160; без дублей.
+ * При нарушениях выходит с кодом 1 (ломает сборку).
+ */
+import fs from 'fs';
+import path from 'path';
+import { fileURLToPath } from 'url';
+
+const __dirname = path.dirname(fileURLToPath(import.meta.url));
+const ROOT = path.resolve(__dirname, '..');
+const BUILD = path.join(ROOT, 'build');
+
+const TITLE_MAX = 70;
+const DESC_MIN = 25;
+const DESC_MAX = 160;
+
+const problems = [];
+const titles = {};
+const descs = {};
+
+function check(id, title, description, { noindex = false } = {}) {
+  if (!title) problems.push(`[${id}] нет <title>`);
+  else if (title.length > TITLE_MAX)
+    problems.push(`[${id}] title ${title.length}>${TITLE_MAX}: ${title}`);
+
+  if (!description) problems.push(`[${id}] нет meta description`);
+  else if (description.length < DESC_MIN || description.length > DESC_MAX)
+    problems.push(
+      `[${id}] description ${description.length} вне ${DESC_MIN}–${DESC_MAX}: ${description}`
+    );
+
+  // дубли (служебные noindex-страницы, напр. 404, не учитываем)
+  if (!noindex) {
+    if (title) (titles[title] = titles[title] || []).push(id);
+    if (description) (descs[description] = descs[description] || []).push(id);
+  }
+}
+
+function walk(dir, filter) {
+  let out = [];
+  for (const name of fs.readdirSync(dir)) {
+    const p = path.join(dir, name);
+    const st = fs.statSync(p);
+    if (st.isDirectory()) out = out.concat(walk(p, filter));
+    else if (filter(p)) out.push(p);
+  }
+  return out;
+}
+
+// Декодирование распространенных HTML сущностей (&amp; -> &) для честной длины
+function decodeHtmlEntities(str) {
+  if (!str) return str;
+  return str
+    .replace(/&amp;/g, '&')
+    .replace(/&lt;/g, '<')
+    .replace(/&gt;/g, '>')
+    .replace(/&quot;/g, '"')
+    .replace(/&#39;/g, "'")
+    .replace(/&nbsp;/g, ' ');
+}
+
+let mode;
+if (fs.existsSync(BUILD)) {
+  mode = 'build HTML';
+  const files = walk(BUILD, (p) => p.endsWith('index.html'));
+  for (const file of files) {
+    const html = fs.readFileSync(file, 'utf8');
+    const t = html.match(/<title[^>]*>([\s\S]*?)<\/title>/i);
+    const d = html.match(
+      /<meta[^>]*name=["']description["'][^>]*content=["']([\s\S]*?)["'][^>]*>/i
+    );
+    const url =
+      '/' +
+      path
+        .relative(BUILD, file)
+        .replace(/index\.html$/, '')
+        .replace(/\/$/, '');
+    const noindex = /name=["']robots["'][^>]*noindex/i.test(html);
+
+    // /publikationen/* являются каноническими алиасами на /publications/*
+    // и делят единый title/description, поэтому исключаем их из проверки дублирования
+    const isAlias = url.startsWith('/publikationen');
+
+    const cleanTitle = t ? decodeHtmlEntities(t[1].trim()) : null;
+    const cleanDesc = d ? decodeHtmlEntities(d[1].trim()) : null;
+
+    check(url || '/', cleanTitle, cleanDesc, { noindex: noindex || isAlias });
+  }
+} else {
+  mode = 'source (build/ не найден)';
+
+  // seoMeta.js
+  const seoPath = path.join(ROOT, 'src/data/seoMeta.js');
+  if (fs.existsSync(seoPath)) {
+    const seoSrc = fs.readFileSync(seoPath, 'utf8');
+    const re =
+      /"(\/[^"]*)":\s*\{\s*\n?\s*title:\s*\n?\s*"((?:[^"\\]|\\.)*)",\s*\n?\s*description:\s*\n?\s*"((?:[^"\\]|\\.)*)"/g;
+    let m;
+    while ((m = re.exec(seoSrc)) !== null) {
+      check(m[1], m[2], m[3], { noindex: m[1] === '/404' });
+    }
+  }
+
+  // articles.js
+  const artPath = path.join(ROOT, 'src/data/articles.js');
+  if (fs.existsSync(artPath)) {
+    const artSrc = fs.readFileSync(artPath, 'utf8');
+    const reA =
+      /"slug":\s*"([^"]+)"[\s\S]*?"seoTitle":\s*"((?:[^"\\]|\\.)*)"[\s\S]*?"seoDescription":\s*"((?:[^"\\]|\\.)*)"/g;
+    let m;
+    while ((m = reA.exec(artSrc)) !== null) {
+      check('article:' + m[1], m[2], m[3]);
+    }
+  }
+}
+
+for (const [t, ids] of Object.entries(titles)) {
+  if (ids.length > 1) problems.push(`Дубль title у ${ids.join(', ')}: ${t}`);
+}
+for (const [d, ids] of Object.entries(descs)) {
+  if (ids.length > 1) problems.push(`Дубль description у ${ids.join(', ')}: ${d}`);
+}
+
+console.log(`\nSEO-аудит (${mode}):`);
+if (problems.length === 0) {
+  console.log('✅ Все title ≤70, description 25–160, дублей нет.\n');
+  process.exit(0);
+} else {
+  console.log(`❌ Найдено проблем: ${problems.length}`);
+  for (const p of problems) console.log('  • ' + p);
+  console.log('');
+  process.exit(1);
+}
